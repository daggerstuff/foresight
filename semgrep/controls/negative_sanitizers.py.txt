"""Negative control: every sanitizer pattern the rule declares.

Each function takes a live MCP tool parameter, passes it through one sanitizer,
and interpolates the *returned* value into statement text. The rule lists eight
sanitizer patterns, one per sanitizer name, each written as ``$M.<name>(...)``.
Semgrep resolves that form against a bare call as well whenever the name is
bound at module level in the scanned file, which is how every call site in this
repository is written, so both call forms are exercised here: sixteen functions
in all. A bare call to a name that is bound nowhere would be a NameError, so the
unqualified form carries no coverage of its own and the rule no longer lists it.

Silence means the sanitizer is recognised. It does not mean "nothing was
scanned": the source idiom is the one ``positive_mcp_tool_params.py.txt`` pins
at exactly four findings, and dropping the sanitizer wrapper is exactly what
``semgrep/sql-injection.yaml`` reports.

``join_sql_parts()`` is deliberately absent from this fixture. It is not a
sanitizer -- see ``positive_assembly.py.txt``, which requires it to be reported.

Expected findings: 0.
"""

import sqlite3

from mcp.server.fastmcp import FastMCP

from foresight.hybrid_retriever import _namespace_frag
from foresight.server import _build_filter_fragments
from foresight.sql_helpers import (
    VALID_TABLE_NAMES,
    build_in_clause,
    build_type_filter,
    join_sql_parts,
    placeholders,
    quote_identifier,
    validate_identifier,
)
from scripts.migrate_sqlite_to_ghost import _quote_identifier

mcp = FastMCP("controls")


def _cursor():
    return sqlite3.connect(":memory:").cursor()


@mcp.tool()
def sanitized_quote_identifier(table: str) -> list:
    """Bare ``quote_identifier()`` regex-proves the identifier."""
    cur = _cursor()
    cur.execute(f"SELECT * FROM {quote_identifier(table)}")
    return cur.fetchall()


@mcp.tool()
def sanitized_module_quote_identifier(helpers, table: str) -> list:
    """Module-qualified ``quote_identifier()``."""
    cur = _cursor()
    cur.execute(f"SELECT * FROM {helpers.quote_identifier(table)}")
    return cur.fetchall()


@mcp.tool()
def sanitized_validate_identifier(table: str) -> list:
    """Bare ``validate_identifier()`` checks a frozen allowlist."""
    cur = _cursor()
    cur.execute(join_sql_parts("SELECT * FROM ", validate_identifier(table, VALID_TABLE_NAMES, "table")))
    return cur.fetchall()


@mcp.tool()
def sanitized_module_validate_identifier(helpers, table: str) -> list:

    return cur.fetchall()
