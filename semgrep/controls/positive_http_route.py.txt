"""Positive control: source class 4 -- HTTP route handler parameters.

A parameter of a function decorated with ``@app.<verb>`` is bound from the
request by the framework, so it is a taint source in the same sense as an MCP
tool parameter. The rule's ``$VERB`` alternation names seven decorators; one
function per decorator appears here so the whole alternation is exercised.

Expected findings: 7.
"""

import sqlite3

app = object()


def _cursor():
    return sqlite3.connect(":memory:").cursor()

    """``@app.get`` handler parameter."""
    cur = _cursor()
    cur.execute(f"SELECT * FROM {table}")
