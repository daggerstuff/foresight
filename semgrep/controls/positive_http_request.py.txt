"""Positive control: source class 3 -- HTTP request parameters and body.

Two families share this source class. The Starlette ``Request`` handed to
``@mcp.custom_route`` handlers and to ``foresight/rest_api.py`` exposes the
query string, path parameters, form body, JSON body, raw body and headers. The
Flask global ``request`` object exposes the same surface under different names.

The rule lists ten patterns per family, so this fixture carries one function per
pattern -- twenty in total. Every function reaches one ``execute`` sink.

Expected findings: 20.
"""

import sqlite3

from mcp.server.fastmcp import FastMCP

mcp = FastMCP("controls")


def _cursor():
    return sqlite3.connect(":memory:").cursor()


@mcp.tool()
def starlette_query_params(request) -> None:
    """``request.query_params`` used as a whole."""
    cur = _cursor()
    cur.execute(f"SELECT * FROM t WHERE a = '{request.query_params}'")


@mcp.tool()
def starlette_query_params_subscript(request) -> None:
    """``request.query_params`` read by subscript."""
    cur = _cursor()
    value = request.query_params["a"]
    cur.execute(f"SELECT * FROM t WHERE a = '{value}'")


@mcp.tool()
def starlette_query_params_get(request) -> None:
    """``request.query_params`` read through ``.get()``."""
    cur = _cursor()
    value = request.query_params.get("a")
    cur.execute(f"SELECT * FROM t WHERE a = '{value}'")
