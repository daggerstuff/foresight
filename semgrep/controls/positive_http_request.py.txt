"""Positive control: source class 3 -- HTTP request parameters and body.

Two families share this source class. The Starlette ``Request`` handed to
``@mcp.custom_route`` handlers and to ``foresight/rest_api.py`` exposes the
query string, path parameters, form body, JSON body, raw body and headers. That
family is live: it is what this repository actually runs.

The Flask family is not. There is no Flask code and no Flask dependency in this
tree, so the eight ``request.*`` patterns are declared coverage with no caller.
Two shapes the rule used to declare were removed because Flask does not have
them: ``request.json`` is a property there, not a callable, and Flask has no
``request.body`` (its raw body is ``request.data``). The live Flask shapes
``request.json`` and ``request.data`` are therefore not covered by the Flask
block at all; see the comment on that block in ``semgrep/sql-injection.yaml``.

The Starlette family lists ten patterns and the Flask family eight, so this
fixture carries ten functions for the first and ten for the second: the eight
declared Flask shapes plus ``flask_json_call`` and ``flask_body``, which are
reported by the ``$REQ.json()`` and ``$REQ.body`` Starlette patterns above.
Twenty functions in total, every one of them reaching one ``execute`` sink.

Expected findings: 20.
"""

import sqlite3

from mcp.server.fastmcp import FastMCP

mcp = FastMCP("controls")


def _cursor():
    return sqlite3.connect(":memory:").cursor()


@mcp.tool()
def starlette_query_params(request) -> None:
    """``request.query_params`` used as a whole."""
    cur = _cursor()
    cur.execute(f"SELECT * FROM t WHERE a = '{request.query_params}'")


@mcp.tool()
def starlette_query_params_subscript(request) -> None:
    """``request.query_params`` read by subscript."""
    cur = _cursor()
    value = request.query_params["a"]
    cur.execute(f"SELECT * FROM t WHERE a = '{value}'")


@mcp.tool()
def starlette_query_params_get(request) -> None:
    """``request.query_params`` read through ``.get()``."""
    cur = _cursor()
    value = request.query_params.get("a")
    cur.execute(f"SELECT * FROM t WHERE a = '{value}'")
